void function(byte &arg){

 * Function with pointer
 */
void function2(byte *arg){

}

     byte *var;
     printf("value: %i\n",*var);
     printf("pointer: %i\n",var);
     {
        function(*var);
        printf("value: %i\n",*var);
     }
     function2(var);
     printf("value: %i\n",*var);
}
